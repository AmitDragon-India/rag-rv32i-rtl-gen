module rv32i_mem_stage (
    input  logic [31:0] alu_result,
    input  logic [31:0] store_data,
    input  logic [31:0] pc_plus4_in,
    input  logic [31:0] imm_u_in,
    input  logic        mem_read,
    input  logic        mem_write,
    input  logic [2:0]  funct3,
    input  logic [1:0]  wb_sel,
    input  logic        reg_write,
    input  logic [4:0]  rd_in,
    output logic [31:0] dmem_addr,
    output logic [31:0] dmem_wdata,
    output logic [3:0]  dmem_byte_en,
    output logic        dmem_we,
    output logic        dmem_re,
    input  logic [31:0] dmem_rdata,
    output logic [31:0] pc_plus4_out,
    output logic [31:0] imm_u_out,
    output logic [31:0] mem_rdata_out,
    output logic [31:0] alu_result_out,
    output logic        reg_write_out,
    output logic [4:0]  rd_out,
    output logic [1:0]  wb_sel_out
);

    // Internal signals
    logic [1:0]  byte_off;
    logic [31:0] load_data;
    logic [7:0]  byte_val;
    logic [15:0] halfword_val;
    logic        sign_bit;

    // Combinational logic for memory access
    always_comb begin
        // Safe defaults for all outputs
        dmem_addr      = 32'h0;
        dmem_wdata     = 32'h0;
        dmem_byte_en   = 4'b0000;
        dmem_we        = 1'b0;
        dmem_re        = 1'b0;
        load_data      = 32'h0;
        byte_off       = alu_result[1:0];

        // Address is word-aligned (upper 30 bits)
        dmem_addr = alu_result;

        // Handle stores
        if (mem_write) begin
            dmem_we = 1'b1;
            
            unique case (funct3)
                3'b000: begin  // SB - Store Byte
                    dmem_byte_en = 4'b0001 << byte_off;
                    dmem_wdata   = store_data << (8 * byte_off);
                end
                3'b001: begin  // SH - Store Halfword
                    dmem_byte_en = byte_off[1] ? 4'b1100 : 4'b0011;
                    dmem_wdata   = store_data << (8 * byte_off);
                end
                3'b010: begin  // SW - Store Word
                    dmem_byte_en = 4'b1111;
                    dmem_wdata   = store_data;
                end
                default: begin
                    dmem_we      = 1'b0;
                    dmem_byte_en = 4'b0000;
                    dmem_wdata   = 32'h0;
                end
            endcase
        end

        // Handle loads
        if (mem_read) begin
            dmem_re = 1'b1;
            
            unique case (funct3)
                3'b000: begin  // LB - Load Byte (sign-extend)
                    byte_val = dmem_rdata >> (8 * byte_off);
                    sign_bit = byte_val[7];
                    load_data = {{24{sign_bit}}, byte_val};
                end
                3'b001: begin  // LH - Load Halfword (sign-extend)
                    halfword_val = dmem_rdata >> (8 * byte_off);
                    sign_bit = halfword_val[15];
                    load_data = {{16{sign_bit}}, halfword_val};
                end
                3'b010: begin  // LW - Load Word
                    load_data = dmem_rdata;
                end
                3'b100: begin  // LBU - Load Byte Unsigned (zero-extend)
                    byte_val = dmem_rdata >> (8 * byte_off);
                    load_data = {24'h0, byte_val};
                end
                3'b101: begin  // LHU - Load Halfword Unsigned (zero-extend)
                    halfword_val = dmem_rdata >> (8 * byte_off);
                    load_data = {16'h0, halfword_val};
                end
                default: begin
                    load_data = 32'h0;
                end
            endcase
        end
    end

    // Combinational logic for pipeline register outputs
    always_comb begin
        // Pass through pipeline registers
        pc_plus4_out   = pc_plus4_in;
        imm_u_out      = imm_u_in;
        mem_rdata_out  = load_data;
        alu_result_out = alu_result;
        reg_write_out  = reg_write;
        rd_out         = rd_in;
        wb_sel_out     = wb_sel;
    end

endmodule
